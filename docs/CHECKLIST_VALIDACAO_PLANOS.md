## 📊 Queries Úteis para Validação

```sql
-- Contar funcionários por empresa
SELECT 
  os.fantasy_name,
  COUNT(p.id) as total_funcionarios
FROM profiles p
JOIN organizational_structure os ON p.root_company_id = os.id
WHERE os.type = 'company' AND os.code LIKE '%-001'
GROUP BY os.fantasy_name;

-- Verificar estrutura organizacional
SELECT 
  code, name, type, 
  (SELECT name FROM organizational_structure p WHERE p.id = os.parent_id) as parent
FROM organizational_structure os
WHERE code LIKE 'CORP-%'
ORDER BY type, code;

-- Listar benefícios por empresa
SELECT 
  os.fantasy_name,
  b.name as beneficio,
  b.benefit_type,
  b.value_per_employee
FROM benefits b
JOIN organizational_structure os ON b.root_company_id = os.id
WHERE os.type = 'company'
ORDER BY os.fantasy_name, b.name;

-- Distribuição salarial por grade
SELECT 
  p.grade,
  COUNT(*) as qtd,
  MIN(p.salary) as min_sal,
  AVG(p.salary) as avg_sal,
  MAX(p.salary) as max_sal
FROM profiles p
JOIN organizational_structure os ON p.root_company_id = os.id
WHERE os.code = 'CORP-001'
GROUP BY p.grade
ORDER BY p.grade;
```

## 🧹 Como Limpar os Dados de Teste

```sql
-- ATENÇÃO: Execute apenas se quiser remover TODOS os dados de teste

-- 1. Remover funcionários
DELETE FROM profiles 
WHERE root_company_id IN (
  SELECT id FROM organizational_structure 
  WHERE code IN ('TECH-001', 'MEDIA-001', 'CORP-001')
);

-- 2. Remover benefícios
DELETE FROM benefits 
WHERE root_company_id IN (
  SELECT id FROM organizational_structure 
  WHERE code IN ('TECH-001', 'MEDIA-001', 'CORP-001')
);

-- 3. Remover faixas salariais
DELETE FROM salary_ranges 
WHERE salary_table_id IN (
  SELECT st.id FROM salary_tables st
  JOIN organizational_structure os ON st.root_company_id = os.id
  WHERE os.code IN ('TECH-001', 'MEDIA-001', 'CORP-001')
);

-- 4. Remover tabelas salariais
DELETE FROM salary_tables 
WHERE root_company_id IN (
  SELECT id FROM organizational_structure 
  WHERE code IN ('TECH-001', 'MEDIA-001', 'CORP-001')
);

-- 5. Remover cargos (se específicos das empresas)
DELETE FROM job_titles 
WHERE code LIKE 'TECH-%' OR code LIKE 'MEDIA-%' OR code LIKE 'CORP-%';

-- 6. Remover estrutura organizacional
DELETE FROM organizational_structure 
WHERE root_company_id IN (
  SELECT id FROM organizational_structure 
  WHERE code IN ('TECH-001', 'MEDIA-001', 'CORP-001')
);

DELETE FROM organizational_structure 
WHERE code IN ('TECH-001', 'MEDIA-001', 'CORP-001');
```
